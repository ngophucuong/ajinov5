## 📁 File `~/ajinov5/worker.js` (Deploy qua Wrangler)

```
// ~/ajinov5/worker.js
// Ajinov5 Agent Gateway - Production Worker
// Domain: ajinov5.cuong.ngo (frontend + auth)
// Tunnel: api.ajinov5.cuong.ngo (VPS: Agno/LiteLLM/Messaging)

import { jwtVerify, SignJWT } from 'https://deno.land/x/jose@v5.2.0/index.ts';

const CONFIG = {
  JWT_SECRET: 'your-production-jwt-secret-min-32-chars-here!!',
  TUNNEL_BASE_URL: 'https://api.ajinov5.cuong.ngo',
  RATE_LIMIT_WINDOW: 60,
  RATE_LIMIT_MAX: 100,
  OTP_TTL: 300,
};

const corsHeaders = {
  'Access-Control-Allow-Origin': 'https://ajinov5.cuong.ngo',
  'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type, Authorization, X-Request-ID',
};

// ─── Rate Limit ───────────────────────────────────────
async function checkRateLimit(env, identifier) {
  const key = 'rl:' + identifier;
  const now = Math.floor(Date.now() / 1000);
  const windowStart = now - CONFIG.RATE_LIMIT_WINDOW;

  let data = await env.RATE_LIMIT_KV.get(key, { type: 'json' }) || { requests: [] };
  data.requests = data.requests.filter(t => t > windowStart);

  if (data.requests.length >= CONFIG.RATE_LIMIT_MAX) {
    return { allowed: false, remaining: 0 };
  }

  data.requests.push(now);
  await env.RATE_LIMIT_KV.put(key, JSON.stringify(data), { expirationTtl: 65 });

  return { allowed: true, remaining: CONFIG.RATE_LIMIT_MAX - data.requests.length };
}

// ─── JWT ──────────────────────────────────────────────
async function verifyJWT(token) {
  try {
    const { payload } = await jwtVerify(token, new TextEncoder().encode(CONFIG.JWT_SECRET), { clockTolerance: 60 });
    return { valid: true, payload };
  } catch { return { valid: false }; }
}

async function createJWT(payload) {
  return new SignJWT(payload)
    .setProtectedHeader({ alg: 'HS256' })
    .setIssuedAt()
    .setExpirationTime('24h')
    .sign(new TextEncoder().encode(CONFIG.JWT_SECRET));
}

// ─── Workers AI Embedding ─────────────────────────────
async function getEmbedding(env, text) {
  const response = await env.AI.run('@cf/baai/bge-m3', { text: [text] });
  return response.data[0];
}

// ─── Vector Search ────────────────────────────────────
async function vectorSearch(env, vector, topK = 5) {
  return await env.VECTORIZE_INDEX.query(vector, { topK });
}

// ─── OTP ──────────────────────────────────────────────
function generateOTP() {
  return Math.floor(100000 + Math.random() * 900000).toString();
}

async function storeOTP(env, identifier, otp) {
  await env.OTP_KV.put('otp:' + identifier, otp, { expirationTtl: CONFIG.OTP_TTL });
}

async function verifyOTP(env, identifier, otp) {
  const stored = await env.OTP_KV.get('otp:' + identifier);
  if (!stored || stored !== otp) return false;
  await env.OTP_KV.delete('otp:' + identifier);
  return true;
}

// ─── Proxy to VPS ─────────────────────────────────────
async function proxyToVPS(request, path) {
  const url = new URL(request.url);
  const target = CONFIG.TUNNEL_BASE_URL + path + url.search;

  const headers = new Headers(request.headers);
  headers.set('X-Forwarded-For', request.headers.get('CF-Connecting-IP') || 'unknown');
  headers.set('X-Request-ID', crypto.randomUUID());

  return fetch(target, {
    method: request.method,
    headers,
    body: request.method !== 'GET' && request.method !== 'HEAD' ? await request.clone().arrayBuffer() : undefined,
  });
}

// ─── Response Helpers ─────────────────────────────────
function json(data, status = 200, extra = {}) {
  return new Response(JSON.stringify(data), {
    status,
    headers: { 'Content-Type': 'application/json', ...corsHeaders, ...extra }
  });
}

function error(status, message) {
  return json({ error: message, status }, status);
}

// ─── Main Handler ─────────────────────────────────────
export default {
  async fetch(request, env, ctx) {
    if (request.method === 'OPTIONS') {
      return new Response(null, { status: 204, headers: corsHeaders });
    }

    const url = new URL(request.url);
    const path = url.pathname;
    const clientIP = request.headers.get('CF-Connecting-IP') || 'unknown';

    try {
      // Health
      if (path === '/health') {
        return json({ status: 'ok', service: 'ajinov5-gateway', time: new Date().toISOString() });
      }

      // ─── Frontend Domain: ajinov5.cuong.ngo ──────────
      if (url.hostname === 'ajinov5.cuong.ngo') {

        // Serve SPA / Telegram Mini App
        if (path === '/') {
          return new Response(`<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Ajinov5 Agent</title>
  <script src="https://telegram.org/js/telegram-web-app.js"></script>
  <style>
    body { font-family: system-ui; max-width: 600px; margin: 0 auto; padding: 20px; }
    .btn { background: #2481cc; color: white; border: none; padding: 12px 24px; border-radius: 8px; cursor: pointer; }
  </style>
</head>
<body>
  <h1>🤖 Ajinov5 Agent</h1>
  <p>Telegram Mini App / Browser Client</p>
  <button class="btn" onclick="sendMessage()">Test Agent</button>
  <div id="output"></div>
  <script>
    const tg = window.Telegram?.WebApp;
    if (tg) tg.ready();

    async function sendMessage() {
      const res = await fetch('/api/health');
      const data = await res.json();
      document.getElementById('output').innerHTML = '<pre>' + JSON.stringify(data, null, 2) + '</pre>';
    }
  </script>
</body>
</html>`, { headers: { 'Content-Type': 'text/html' } });
        }

        // Auth: OTP request
        if (path === '/auth/otp' && request.method === 'POST') {
          const body = await request.json();
          const otp = generateOTP();
          await storeOTP(env, body.identifier, otp);
          return json({ success: true, message: 'OTP sent', dev_otp: otp });
        }

        // Auth: OTP verify
        if (path === '/auth/verify' && request.method === 'POST') {
          const body = await request.json();
          const valid = await verifyOTP(env, body.identifier, body.otp);
          if (!valid) return error(401, 'Invalid OTP');
          const token = await createJWT({ sub: body.identifier });
          return json({ success: true, token });
        }

        // Proxy API to VPS
        if (path.startsWith('/api/')) {
          const rate = await checkRateLimit(env, clientIP);
          if (!rate.allowed) return error(429, 'Rate limited');

          const response = await proxyToVPS(request, path.replace('/api', ''));
          const headers = new Headers(response.headers);
          Object.entries(corsHeaders).forEach(([k,v]) => headers.set(k,v));
          headers.set('X-RateLimit-Remaining', rate.remaining.toString());
          return new Response(response.body, { status: response.status, headers });
        }
      }

      // ─── API Domain: api.ajinov5.cuong.ngo ───────────
      // This receives direct requests proxied through Tunnel to VPS
      if (url.hostname === 'api.ajinov5.cuong.ngo') {
        return json({ 
          status: 'ok', 
          message: 'Direct API access via Tunnel',
          path,
          note: 'Requests are proxied to VPS:8000 (Agno Runtime)'
        });
      }

      return error(404, 'Not found');

    } catch (err) {
      console.error('Error:', err);
      return error(500, err.message);
    }
  }
};
```

## 🔧 Deploy Worker qua Wrangler CLI

```
# 1. Install Wrangler
npm install -g wrangler

# 2. Login
wrangler login

# 3. Set secrets
wrangler secret put JWT_SECRET --name agent-gateway
# Enter: your-32-char-minimum-secret-key

# 4. Deploy
cd ~/ajinov5
wrangler deploy

# 5. Add custom domain
wrangler route add "ajinov5.cuong.ngo/*" agent-gateway
```

## 🔧 Cloudflare Tunnel Setup

```
# 1. Install cloudflared
wget -q https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-amd64.deb
sudo dpkg -i cloudflared-linux-amd64.deb

# 2. Login (opens browser)
cloudflared tunnel login

# 3. Create tunnel
cloudflared tunnel create ajinov5-vps
# → Save the Tunnel ID and credentials file path

# 4. Get token for Docker
cloudflared tunnel token ajinov5-vps
# → Copy token to .env TUNNEL_TOKEN

# 5. Configure DNS routes
cloudflared tunnel route dns ajinov5-vps api.ajinov5.cuong.ngo
cloudflared tunnel route dns ajinov5-vps webhook.ajinov5.cuong.ngo

# 6. Config file ~/.cloudflared/config.yml
echo "tunnel: <TUNNEL_ID>
credentials-file: /root/.cloudflared/<TUNNEL_ID>.json
ingress:
  - hostname: api.ajinov5.cuong.ngo
    service: http://localhost:8000
  - hostname: webhook.ajinov5.cuong.ngo
    service: http://localhost:3000
  - service: http_status:404" > ~/.cloudflared/config.yml

# 7. Run tunnel (or use Docker in compose)
cloudflared tunnel run ajinov5-vps
```
