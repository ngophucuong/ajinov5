### Deploy commands
```bash
# Frontend + Worker
cd apps/web && npm run build
rm -rf ../worker/static/* && cp -r dist/* ../worker/static/
cd ../worker
JSFILE=$(ls static/assets/index-*.js | head -1 | xargs basename)
CSSFILE=$(ls static/assets/index-*.css | head -1 | xargs basename)
perl -i -pe "s|/assets/index-[A-Za-z0-9\-]+\.js|/assets/$JSFILE|g" src/index.ts
perl -i -pe "s|/assets/index-[A-Za-z0-9\-]+\.css|/assets/$CSSFILE|g" src/index.ts
wrangler deploy

# VPS backend
scp services/agno/main.py root@72.60.210.110:~/ajinov5/agno/
ssh root@72.60.210.110 'cd ~/ajinov5 && docker compose restart agno'
```
