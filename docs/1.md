## 1. VPS Docker Compose

Tạo thư mục `~/ajinov5` và file `docker-compose.yml`:

```
# ~/ajinov5/docker-compose.yml
version: "3.8"

services:
  # ─── PostgreSQL + pgvector ─────────────────────────────
  postgres:
    image: ankane/pgvector:v0.5.1
    container_name: ajinov5-postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: ajinov5
      POSTGRES_USER: ajinov5
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-changeme-strong-password}
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./backups:/backups
    ports:
      - "127.0.0.1:5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ajinov5 -d ajinov5"]
      interval: 10s
      timeout: 5s
      retries: 5

  # ─── Agno Runtime (Orchestration + Skills) ─────────────
  agno:
    image: python:3.11-slim
    container_name: ajinov5-agno
    restart: unless-stopped
    working_dir: /app
    environment:
      - DATABASE_URL=postgresql://ajinov5:${POSTGRES_PASSWORD:-changeme-strong-password}@postgres:5432/ajinov5
      - OPENAI_API_KEY=${OPENAI_API_KEY}
      - DEEPSEEK_API_KEY=${DEEPSEEK_API_KEY}
      - LITELLM_API_BASE=http://litellm:4000
      - SERPER_API_KEY=${SERPER_API_KEY}
      - EMBEDDING_API_BASE=http://localhost:8000/embed  # Self-call via Workers
    volumes:
      - ./agno:/app
      - ./backups:/app/backups
    ports:
      - "127.0.0.1:8000:8000"
    depends_on:
      postgres:
        condition: service_healthy
    command: >
      sh -c "pip install agno fastapi uvicorn psycopg2-binary pgvector httpx &&
             uvicorn main:app --host 0.0.0.0 --port 8000 --reload"
    # Production: build custom image with requirements.txt

  # ─── LiteLLM (LLM Router + Cost Tracking) ──────────────
  litellm:
    image: ghcr.io/berriai/litellm:main-latest
    container_name: ajinov5-litellm
    restart: unless-stopped
    environment:
      - LITELLM_MASTER_KEY=${LITELLM_MASTER_KEY:-sk-litellm-master}
      - DEEPSEEK_API_KEY=${DEEPSEEK_API_KEY}
      - OPENAI_API_KEY=${OPENAI_API_KEY}
    volumes:
      - ./litellm-config.yaml:/app/config.yaml:ro
    ports:
      - "127.0.0.1:4000:4000"
    command: >
      --config /app/config.yaml
      --port 4000

  # ─── Messaging Gateway (Telegram + WhatsApp) ────────────
  messaging:
    image: node:20-alpine
    container_name: ajinov5-messaging
    restart: unless-stopped
    working_dir: /app
    environment:
      - TELEGRAM_BOT_TOKEN=${TELEGRAM_BOT_TOKEN}
      - WHATSAPP_API_KEY=${WHATSAPP_API_KEY}
      - WHATSAPP_PHONE_ID=${WHATSAPP_PHONE_ID}
      - WEBHOOK_URL=https://api.ajinov5.cuong.ngo/webhook
      - AGNO_API_URL=http://agno:8000
    volumes:
      - ./messaging:/app
    ports:
      - "127.0.0.1:3000:3000"
    command: >
      sh -c "npm install express body-parser axios node-telegram-bot-api &&
             node server.js"

  # ─── Cloudflared Tunnel ────────────────────────────────
  tunnel:
    image: cloudflare/cloudflared:latest
    container_name: ajinov5-tunnel
    restart: unless-stopped
    command: tunnel run --token ${TUNNEL_TOKEN}
    # Token from: cloudflared tunnel create ajinov5-vps
```

### `~/ajinov5/messaging/server.js` (starter)

```
const express = require('express');
const bodyParser = require('body-parser');
const axios = require('axios');

const app = express();
app.use(bodyParser.json());

const TELEGRAM_TOKEN = process.env.TELEGRAM_BOT_TOKEN;
const AGNO_URL = process.env.AGNO_API_URL || 'http://agno:8000';
const WEBHOOK_URL = process.env.WEBHOOK_URL;

// Telegram webhook handler
app.post('/webhook/telegram', async (req, res) => {
    const { message } = req.body;
    if (!message) return res.sendStatus(200);

    const chatId = message.chat.id;
    const text = message.text;

    try {
        // Forward to Agno Runtime
        const response = await axios.post(`${AGNO_URL}/agent/run`, {
            user_id: chatId,
            message: text,
            source: 'telegram'
        });

        // Reply to user
        await axios.post(`https://api.telegram.org/bot${TELEGRAM_TOKEN}/sendMessage`, {
            chat_id: chatId,
            text: response.data.response || 'Processing...'
        });
    } catch (err) {
        console.error('Error:', err.message);
    }

    res.sendStatus(200);
});

// Health check
app.get('/health', (req, res) => {
    res.json({ status: 'ok', service: 'messaging-gateway' });
});

app.listen(3000, '0.0.0.0', () => {
    console.log('Messaging Gateway on :3000');
    console.log('Webhook URL:', WEBHOOK_URL + '/webhook/telegram');
});
```
